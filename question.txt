class Service:

    def __init__(self,service_name: str,resolver_authority: str):

        self.resolver_authority = resolver_authority

        self.name = service_name

        self.server = server.serve(self.name,resolver_authority=self.resolver_authority)

        self.thread = None

    def start_run(self):

        print(f"starting service {self.name} ...")

        self.server.start()

        print(f"service {self.name} started")

        self.thread = Thread(target=self.server.wait_for_termination)

        self.thread.start()


    def stop_run(self):

        print(f"stopping service {self.name} ...")

        shutdwon_event = self.server.stop(grace=10)

        shutdwon_event.wait()

        if not self.thread is None:

            self.thread.join()

        print(f"service {self.name} stoped")
